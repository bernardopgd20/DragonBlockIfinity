package com.bernardo.dbi.core;

import net.minecraft.core.BlockPos;
import net.minecraft.world.entity.Entity;

public abstract class SafeZone {

    protected BlockPos center;
    protected int radiusX;
    protected int radiusY;
    protected int radiusZ;

    protected SafeZone(BlockPos center, int radiusX, int radiusY, int radiusZ) {
        this.center = center;
        this.radiusX = radiusX;
        this.radiusY = radiusY;
        this.radiusZ = radiusZ;
    }

    public boolean isInside(BlockPos pos) {
        return Math.abs(pos.getX() - center.getX()) <= radiusX
            && Math.abs(pos.getY() - center.getY()) <= radiusY
            && Math.abs(pos.getZ() - center.getZ()) <= radiusZ;
    }

    public boolean isInside(Entity entity) {
        return isInside(entity.blockPosition());
    }
}

package com.bernardo.dbi.core;

import com.mojang.blaze3d.systems.RenderSystem;
import com.mojang.blaze3d.vertex.PoseStack;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.renderer.entity.EntityRenderDispatcher;
import net.minecraft.world.entity.player.Player;
import org.joml.Matrix4f;
import org.joml.Quaternionf;

public class PlayerPreview {

    public static void render(GuiGraphics graphics, int x, int y, int size, float yaw, float pitch) {
        Minecraft mc = Minecraft.getInstance();
        Player player = mc.player;
        if (player == null) return;

        PoseStack poseStack = graphics.pose();
        poseStack.pushPose();
        poseStack.translate(x, y, 50);
        poseStack.scale(size, size, -size);

        Quaternionf rot = new Quaternionf().rotateZ((float) Math.PI);
        Quaternionf pitchRot = new Quaternionf().rotateX(pitch * (float)(Math.PI / 180));
        rot.mul(pitchRot);
        poseStack.mulPose(rot);

        float savedYBodyRot = player.yBodyRot;
        float savedYRot = player.getYRot();
        float savedXRot = player.getXRot();
        float savedYHeadRotO = player.yHeadRotO;
        float savedYHeadRot = player.yHeadRot;

        player.yBodyRot = 180f + yaw;
        player.setYRot(180f + yaw);
        player.setXRot(-pitch);
        player.yHeadRot = player.getYRot();
        player.yHeadRotO = player.getYRot();

        EntityRenderDispatcher dispatcher = mc.getEntityRenderDispatcher();
        dispatcher.setRenderShadow(false);
        RenderSystem.runAsFancy(() -> {
            dispatcher.render(player, 0, 0, 0, 0f, 1f,
                poseStack, graphics.bufferSource(), 0xF000F0);
        });
        graphics.flush();
        dispatcher.setRenderShadow(true);

        player.yBodyRot = savedYBodyRot;
        player.setYRot(savedYRot);
        player.setXRot(savedXRot);
        player.yHeadRotO = savedYHeadRotO;
        player.yHeadRot = savedYHeadRot;

        poseStack.popPose();
    }
}

=== src/main/java/com/bernardo/dbi/client/render/aura/AuraManager.java ===
package com.bernardo.dbi.client.render.aura;

import com.bernardo.dbi.entity.AuraEntity;
import com.bernardo.dbi.entity.DBIEntities;
import net.minecraft.world.entity.player.Player;

import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public class AuraManager {

    private static final Map<UUID, AuraEntity> AURAS = new HashMap<>();

    public static void showAura(Player player, float r, float g, float b) {
        UUID id = player.getUUID();
        AuraEntity existing = AURAS.get(id);

        // Se a entidade foi removida do mundo, recria
        if (existing != null && (existing.isRemoved() || existing.isInvisible())) {
            existing.discard();
            AURAS.remove(id);
            existing = null;
        }

        if (existing != null) {
            existing.setColor(r, g, b);
            existing.setInvisible(false);
            return;
        }

        AuraEntity aura = new AuraEntity(DBIEntities.AURA.get(), player.level());
        aura.setColor(r, g, b);
        aura.setInvisible(false);
        aura.moveTo(player.getX(), player.getY(), player.getZ());
        player.level().addFreshEntity(aura);
        AURAS.put(id, aura);
    }

    public static void hideAura(Player player) {
        UUID id = player.getUUID();
        AuraEntity aura = AURAS.remove(id);
        if (aura != null) aura.discard();
    }

    public static void tickAura(Player player) {
        AuraEntity aura = AURAS.get(player.getUUID());
        if (aura == null || aura.isRemoved()) return;
        aura.moveTo(player.getX(), player.getY(), player.getZ());
        aura.setYRot(player.getYRot());
        aura.setInvisible(false);
    }

    public static void ensureVisible() {
        for (AuraEntity aura : AURAS.values()) {
            if (aura.isInvisible()) aura.setInvisible(false);
        }
    }

    public static void clearAll() {
        for (AuraEntity aura : AURAS.values()) aura.discard();
        AURAS.clear();
    }
}

package com.bernardo.dbi.client.menu;

import net.minecraft.client.gui.screens.Screen;
import net.minecraft.network.chat.Component;

public abstract class DBIScreen extends Screen {

    protected float ratioX;
    protected float ratioY;
    protected int guiWidth;
    protected int guiHeight;
    protected int guiLeft;
    protected int guiTop;

    protected DBIScreen(Component title) {
        super(title);
    }

    @Override
    protected void init() {
        ratioX = this.width / 320.0f;
        ratioY = this.height / 240.0f;
        float ratio = Math.min(ratioX, ratioY) * 0.75f;

        guiWidth  = (int)(256 * ratio);
        guiHeight = (int)(192 * ratio);
        guiLeft   = (this.width  - guiWidth)  / 2;
        guiTop    = (this.height - guiHeight) / 2;

        initGui();
    }

    // Cada menu implementa aqui no lugar do init()
    protected abstract void initGui();

    protected int scaled(int base) {
        float ratio = Math.min(ratioX, ratioY) * 0.75f;
        return (int)(base * ratio);
    }
}

import com.bernardo.dbi.client.menu.DBIScreen;
import com.bernardo.dbi.client.menu.MenuRouter;
import com.bernardo.dbi.core.PlayerPreview;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.network.chat.Component;

public class CharacterCreationScreen extends DBIScreen {

    private float yaw = 0f;
    private float pitch = -10f;
    private boolean dragging = false;
    private double lastMouseX, lastMouseY;

    public CharacterCreationScreen() {
        super(Component.translatable("screen.dragonblockinfinity.character_creation"));
    }

    @Override
    protected void initGui() {}

    @Override
    public void render(GuiGraphics graphics, int mouseX, int mouseY, float partialTick) {
        renderBackground(graphics);

        // Fundo do menu
        graphics.fill(guiLeft, guiTop, guiLeft + guiWidth, guiTop + guiHeight, 0xCC000000);

        // Player model centralizado na metade esquerda do menu
        int previewX = guiLeft + guiWidth / 4;
        int previewY = guiTop + guiHeight / 2 + scaled(40);
        int previewSize = scaled(30);

        PlayerPreview.render(graphics, previewX, previewY, previewSize, yaw, pitch);

        super.render(graphics, mouseX, mouseY, partialTick);
    }

    @Override
    public boolean mouseClicked(double mouseX, double mouseY, int button) {
        dragging = true;
        lastMouseX = mouseX;
        lastMouseY = mouseY;
        return super.mouseClicked(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseReleased(double mouseX, double mouseY, int button) {
        dragging = false;
        return super.mouseReleased(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseDragged(double mouseX, double mouseY, int button, double dragX, double dragY) {
        if (dragging) {
            yaw   += (float)(mouseX - lastMouseX) * 1.5f;
            pitch += (float)(mouseY - lastMouseY) * 1.0f;
            pitch  = Math.max(-30f, Math.min(30f, pitch));
            lastMouseX = mouseX;
            lastMouseY = mouseY;
        }
        return super.mouseDragged(mouseX, mouseY, button, dragX, dragY);
    }

    @Override
    public boolean isPauseScreen() { return false; }
}

        new ResourceLocation(DragonBlockInfinity.MOD_ID, "textures/gui/icons_btn.png");
